  String id,
  DateTime at, {
  String tier = 'candidate',

    NightlyHighlight(
      clipId: id,

              batchId: dayKey,
              status: 'ready',
              captureStart: parseDayKey(dayKey)!.add(const Duration(hours: 22)),
              captureEnd: parseDayKey(dayKey)!.add(const Duration(hours: 30)),
              publishedAt: parseDayKey(dayKey)!.add(const Duration(hours: 31))),
      startedAt: at,
      source: 'rule',
      reason: '움직임 3.0초',
      tier: tier,
      dayKey: dayKey,
      playFromSec: playFrom,
      episodeRank: rank,
      episodeClipCount: 4,
      episodeActivitySec: 42,
    );
